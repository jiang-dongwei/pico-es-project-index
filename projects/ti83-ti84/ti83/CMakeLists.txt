cmake_minimum_required(VERSION 3.20)
project(ti83_host C)
option(TI83_CPU_PROFILE "Count host instruction categories" OFF)
option(TI83_TIMING_PROFILE "Sample CPU instruction timings" OFF)
add_compile_definitions(TI83_TIMING_PROFILE=$<BOOL:${TI83_TIMING_PROFILE}>)
if(TI83_CPU_PROFILE)
    add_compile_definitions(TI83_CPU_PROFILE=1)
endif()
set(CMAKE_C_STANDARD 11)
include(core/sources.cmake)
add_library(ti83_core STATIC ${TI83_CORE_SOURCES})
target_include_directories(ti83_core PUBLIC core upstream/emu)
set_source_files_properties(core/adapter.c core/allocator.c core/program.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
set_source_files_properties(core/fast_memory.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
set_source_files_properties(${TI83_CPU_SOURCE} PROPERTIES COMPILE_OPTIONS "-Os;-include;${TI83_ROOT}/core/pico_hot.h")
add_library(ti83_cpu_reference STATIC "${CMAKE_CURRENT_BINARY_DIR}/ti83_cpu/z80_reference.c"
    "${CMAKE_CURRENT_BINARY_DIR}/ti83_cpu/keypad_reference.c")
target_include_directories(ti83_cpu_reference PRIVATE upstream/emu)
enable_testing()
add_executable(test_ti83 tests/test_core.c)
target_link_libraries(test_ti83 ti83_core)
target_compile_options(test_ti83 PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME ti83_core COMMAND test_ti83)
add_executable(test_frontend tests/test_frontend.c
    platform/retropicoes/frontend.c
    ../common/menu/menu_contract.c ../common/menu/menu_input.c ../common/menu/menu_render.c
    ../common/menu/menu_state.c)
target_include_directories(test_frontend PRIVATE platform/retropicoes
    ../compat ../common/menu ../digger/platform/retropicoes)
target_link_libraries(test_frontend ti83_core)
target_compile_options(test_frontend PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME ti83_frontend COMMAND test_frontend)
add_executable(rom_smoke tests/rom_smoke.c)
target_link_libraries(rom_smoke ti83_core)
target_compile_options(rom_smoke PRIVATE -Wall -Wextra -Werror)
add_executable(game_smoke tests/game_smoke.c)
target_link_libraries(game_smoke ti83_core ti83_cpu_reference)
target_compile_options(game_smoke PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_executable(test_program tests/test_program.c)
target_link_libraries(test_program ti83_core)
target_compile_options(test_program PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME ti83_program COMMAND test_program)

    platform/retropicoes/frontend.c
    ../common/menu/menu_contract.c ../common/menu/menu_input.c ../common/menu/menu_render.c
    ../common/menu/menu_state.c)
target_include_directories(frontend_smoke PRIVATE platform/retropicoes
    ../compat ../common/menu ../digger/platform/retropicoes)
target_link_libraries(frontend_smoke ti83_core)
target_compile_options(frontend_smoke PRIVATE -Wall -Wextra -Werror -UNDEBUG)

target_link_libraries(test_fast_memory ti83_core)
target_compile_options(test_fast_memory PRIVATE -Wall -Wextra -Werror -UNDEBUG)
