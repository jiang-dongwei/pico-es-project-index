cmake_minimum_required(VERSION 3.20)
project(studio2_host C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_library(studio2_core core/cdp1802.c core/studio2.c platform/retropicoes/path_adapter.c platform/retropicoes/settings_adapter.c platform/retropicoes/menu_adapter.c)
target_include_directories(studio2_core PUBLIC core platform/retropicoes ../common/menu ../compat)
add_subdirectory(../common/frontend common-frontend)
target_link_libraries(studio2_core PUBLIC retropicoes_frontend)
add_executable(studio2_tests tests/test_studio2.c)
target_link_libraries(studio2_tests PRIVATE studio2_core)
foreach(t studio2_core studio2_tests)
 target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endforeach()
add_test(NAME studio2_tests COMMAND studio2_tests)

add_executable(studio2_tennis_probe tests/probe_tennis.c)
target_link_libraries(studio2_tennis_probe PRIVATE studio2_core)
target_compile_options(studio2_tennis_probe PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)

add_executable(studio2_frontend_tests tests/test_frontend.c platform/retropicoes/frontend.c ../compat/audio.c)
set(RETROPICOES_PLATFORM_BASE "E:/ComporyProject/Pico_ES/digger" CACHE PATH "Read-only board backend")
target_include_directories(studio2_frontend_tests PRIVATE "${RETROPICOES_PLATFORM_BASE}/platform/retropicoes")
target_link_libraries(studio2_frontend_tests PRIVATE studio2_core)
target_compile_options(studio2_frontend_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME studio2_frontend_tests COMMAND studio2_frontend_tests)
